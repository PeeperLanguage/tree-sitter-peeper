fn inspect(result: Result<i32>) {
    match result {
        Result<i32>::Ok with .{ value as payload, _ } => {
            use(payload);
        }
        Result<i32>::Error => {
            report(result.message);
        }
        Result<i32>::Pending => {
            wait();
        }
    }
}

fn read(slot: Slot) -> i32 {
    match &mut holder.slot {
        Slot::Value with none => { return 0; }

        Slot::Inner with Shape::Circle with .{ radius, scale as factor } => { return radius; }

    }

fn take(pair: Pair, maybe: ?i32) -> i32 {
    let .{ first as head, tag, _ } = pair;
    let count = maybe orelse 0;
    let item = find(head) orelse { return tag; };
    return add(count, next() orelse item orelse 1);
}

fn read(shape: Shape, maybe: ?i32) -> i32 {
    if &shape is Shape::Circle { return shape.radius; }
    if !(shape is Shape::Code) { return 0; }
    for maybe != none { return maybe; }
    return shape;
}
