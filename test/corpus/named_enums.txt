fn inspect(result: Result<i32>) {
    match result {
        Result<i32>::Ok{ value = payload } => {
            use(payload);
        }
        Result<i32>::Error{ message = _ } => {
            report();
        }
        Result<i32>::Pending => {
            wait();
        }
    }
}
